\begin{definition}[Monotone Classifier]
  A classifier $h : \R^d \to \{-1,1\}$ is \emph{monotone} if
  $h(p) \ge h(q)$ for every $p,q \in \R^d$ satisfying $p \dom q$.
\end{definition}

\begin{problem}[Monotone Classification with Relative Precision]\label{prob:main}
  Given $\epsilon \ge 0$, find a monotone classifier whose error on $P$
  is at most $(1+\epsilon)\kstar$.
  The efficiency of an algorithm is measured by the number of elements probed.
\end{problem}

\section{Problem Definitions}
% ============================================================

\subsection{Math Conventions}

For an integer $x \ge 1$, the notation $[x]$ denotes $\{1, 2, \ldots, x\}$.
We write $x \le_{1+\epsilon} y$ to mean $x \le (1+\epsilon)y$.
For any real $x \ge 1$, define $\mathrm{Log}\, x = \log_2(1+x)$.

\subsection{Dominance}

A point $p \in \R^d$ \emph{dominates} $q \in \R^d$ (written $p \dom q$) if
$p \ne q$ and $p[i] \ge q[i]$ for all $i \in [d]$.

% ── Inline numbered equation ──
The optimal monotone error is
\begin{equation}\label{eq:kstar}
  \kstar \;=\; \min_{h \in \Hmon} \err_P(h).
\end{equation}

% ── Aligned multi-line equation ──
The expected cost and error of a randomised algorithm $\mathcal{A}$ are
\begin{align}
  \text{expected cost}  &= \E[X], \label{eq:cost}\\
  \text{expected error} &= \E[\err_P(h)]. \label{eq:error}
\end{align}
